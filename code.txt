// ---------- Buzzer ----------
unsigned long buzzTimer = 0;
bool buzzOnPhase = false;
const unsigned long BZ_ON  = 180;
const unsigned long BZ_OFF = 650;


struct SensorData {
  int id;
  String name;
  int quantity;
  String status;
};


SensorData sensors[6] = {
  {1, "smoke", 0, "green"},
  {1, "methane", 0, "green"},
  {1, "co", 0, "green"},
  {2, "smoke", 0, "green"},

};

void updateLCD() {
  lcd.clear();
  lcd.setCursor(0,0);
  lcd.print("U1:");
  
  // وضعیت واحد 1
  bool unit1Alarm = false;
  for(int i = 0; i < 3; i++) {
    if(sensors[i].status == "red") {
      unit1Alarm = true;
      break;
    }
  }
  lcd.print(unit1Alarm ? "ALARM" : "OK");
  
  lcd.setCursor(0,1);
  lcd.print("U2:");
  
  // وضعیت واحد 2
  bool unit2Alarm = false;
  for(int i = 3; i < 6; i++) {
    if(sensors[i].status == "red") {
      unit2Alarm = true;
      break;
    }
  }
  lcd.print(unit2Alarm ? "ALARM" : "OK");
}

void buzzerService(){
  if (!anyAlarm) { 
    noTone(PIN_BUZZ); 
    noTone(PIEZO_PIN);
    buzzOnPhase=false; 
    return; 
  }
  unsigned long now = millis();
  if (!buzzOnPhase && now - buzzTimer >= BZ_OFF) {
    tone(PIEZO_PIN, 1500);
    buzzOnPhase = true;
    buzzTimer = now;
  } else if (buzzOnPhase && now - buzzTimer >= BZ_ON) {
    noTone(PIEZO_PIN);
    buzzOnPhase = false;
    buzzTimer = now;
  }
}

void applyLed(){
  if (!loraOk) {
    digitalWrite(LED_Y, HIGH);
    digitalWrite(LED_G, LOW);
    digitalWrite(LED_R, LOW);
    return;
  }
  digitalWrite(LED_Y, LOW);
  if (anyAlarm){
    digitalWrite(LED_R, HIGH);
  } else {
    digitalWrite(LED_R, LOW);
  }
}

Hazard parseHazard(const String &s){
  if (s == "SMOKE")   return HZ_SMOKE;
  if (s == "METHANE") return HZ_METHANE;
  if (s == "CO")      return HZ_CO;
  return HZ_SMOKE;
}

String hazardToSensorName(Hazard hz) {
  switch(hz) {
    case HZ_SMOKE: return "smoke";
    case HZ_METHANE: return "methane";
    case HZ_CO: return "co";
    default: return "";
  }
}

void updateSensorData(uint8_t unit, Hazard hz, bool isActive) {
  String sensorName = hazardToSensorName(hz);
  
  for(int i = 0; i < 6; i++) {
    if(sensors[i].id == unit && sensors[i].name == sensorName) {
      sensors[i].quantity = isActive ? 1 : 0;
      sensors[i].status = isActive ? "red" : "green";
      
      if (!isActive) {
        lastClearTime[unit][hz] = millis();
      }
      break;
    }
  }
  updateLCD(); 
}

String getSensorStatusWithWarning(uint8_t unit, Hazard hz) {
  int sensorIndex = (unit - 1) * 3 + hz;
  if (sensors[sensorIndex].status == "green") {
    if (millis() - lastClearTime[unit][hz] < 10000) {
      return "orange";
    }
  }
  return sensors[sensorIndex].status;
}

void processMessage(const String &msg){
  int p1 = msg.indexOf(':');
  int p2 = msg.indexOf(':', p1+1);
  if (p1<0 || p2<0) return;

  String prefix = msg.substring(0, p1);         
  String kind   = msg.substring(p1+1, p2);      
  String type   = msg.substring(p2+1);          

  uint8_t unit = 0;
  if (prefix.startsWith("UNIT")) {
    unit = (uint8_t)prefix.substring(4).toInt();
  }
  if (unit < 1 || unit > 2) return;

  Hazard hz = parseHazard(type);

  if (kind == "ALARM") {
    active[unit][hz] = true;
    lastAlarmTime[unit][hz] = millis();
    updateSensorData(unit, hz, true);

    if (hz == HZ_METHANE) {
      digitalWrite(LED_G, HIGH);
    }

  } else if (kind == "CLEAR") {
    active[unit][hz] = false;
    lastAlarmTime[unit][hz] = millis();
    updateSensorData(unit, hz, false);

    if (hz == HZ_METHANE) {
      digitalWrite(LED_G, LOW);
    }
  }

  anyAlarm = false;
  for (uint8_t u=1; u<=2; u++){
    for (int h=0; h<HZ_COUNT; h++){
      if (active[u][h]) { anyAlarm = true; break; }
    }
    if (anyAlarm) break;
  }

  applyLed();
}

void handleSensors() {
  DynamicJsonDocument doc(1024);
  JsonArray array = doc.to<JsonArray>();
  
  unsigned long now = millis();
  
  for(int i = 0; i < 6; i++) {
    JsonObject sensor = array.createNestedObject();
    sensor["id"] = sensors[i].id;
    sensor["name"] = sensors[i].name;
    sensor["quantity"] = sensors[i].quantity;
    

    if (sensors[i].status == "green") {
      // پیدا کردن نوع Hazard از نام سنسور
      Hazard hz;
      if (sensors[i].name == "smoke") hz = HZ_SMOKE;
      else if (sensors[i].name == "methane") hz = HZ_METHANE;
      else if (sensors[i].name == "co") hz = HZ_CO;
      else hz = HZ_SMOKE;
      

      if (now - lastClearTime[sensors[i].id][hz] < 10000) {
        sensor["status"] = "orange";
      } else {
        sensor["status"] = "green";
      }
    } else {
      sensor["status"] = sensors[i].status;
    }
  }
  
  String response;
  serializeJson(doc, response);
  server.send(200, "application/json", response);
}

// ---------- Setup ----------
void setup(){
  pinMode(PIN_BUZZ, OUTPUT);
  pinMode(PIEZO_PIN, OUTPUT);
  noTone(PIN_BUZZ);
  noTone(PIEZO_PIN);

  pinMode(LED_R, OUTPUT);
  pinMode(LED_Y, OUTPUT);
  pinMode(LED_G, OUTPUT);
  digitalWrite(LED_R, LOW);
  digitalWrite(LED_Y, LOW);
  digitalWrite(LED_G, LOW);

  lcd.init();
  lcd.backlight();
  lcd.clear();
  lcd.setCursor(0,0); lcd.print("ImenYar Security");
  lcd.setCursor(3,1); lcd.print("System Starting.");
  delay(1300);

  lcd.clear();
  lcd.setCursor(0,0); lcd.print("Checking LoRa...");
  digitalWrite(LED_Y, HIGH);

  LoRa.setPins(LORA_SS, LORA_RST, LORA_DIO0);
  loraOk = LoRa.begin(433E6);
  if (!loraOk) {
    lcd.clear();
    lcd.setCursor(0,0); lcd.print("LoRa FAILED");
    lcd.setCursor(0,1); lcd.print("check wiring   ");
    applyLed();
    for(;;) { buzzerService(); delay(10); }
  }

  lcd.clear();
  lcd.setCursor(0,0); lcd.print("LoRa OK  (433)");
  lcd.setCursor(0,1); lcd.print("central online ");
  digitalWrite(LED_Y, LOW);
  digitalWrite(LED_G, LOW);
  delay(900);

  lcd.clear();
  updateLCD();
  applyLed();


  for (uint8_t u=0; u<3; u++) {
    for (int h=0; h<HZ_COUNT; h++){
      active[u][h] = false;
      lastAlarmTime[u][h] = 0;
      lastClearTime[u][h] = 0;
    }
  }

  // WiFi AP
  WiFi.softAP(ap_ssid, ap_password);
  Serial.begin(115200);
  Serial.println("Access Point Started");
  Serial.print("AP IP: "); Serial.println(WiFi.softAPIP());

  // WebServer
  server.on("/", handleRoot);
  server.on("/api/sensors", handleSensors);
  server.begin();
}

// ---------- Loop ----------
void loop(){
  int packetSize = LoRa.parsePacket();
  if (packetSize){
    String msg;
    while (LoRa.available()) msg += (char)LoRa.read();
    processMessage(msg);
  }

  buzzerService();
  server.handleClient();
  delay(10);
}
